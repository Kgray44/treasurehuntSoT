import fs from "node:fs";
import assert from "node:assert/strict";
import crypto from "node:crypto";
import { spawnSync } from "node:child_process";
const scratch = "D:/CodexTemp/parallax-resume-20261021-01a15554";
const mode = process.argv[2];
assert(["focused", "typecheck", "lint"].includes(mode));
const freezeBytes = fs.readFileSync(`${scratch}/lint-product-fix-01/source-freeze.original.json`);
const freeze = JSON.parse(freezeBytes);
const hash = (bytes) => crypto.createHash("sha256").update(bytes).digest("hex");
assert.equal(hash(freezeBytes), "1e8fdba87ed5527a8bbfa29d2e9d70b9bf780018847981201249f7c85106835e");
for (const [file, expected] of Object.entries(freeze.source)) assert.equal(hash(fs.readFileSync(file)), expected, file);

if (mode === "typecheck") args = ["node_modules/typescript/bin/tsc", "--noEmit"];
if (mode === "lint") {
  const planBytes = fs.readFileSync(`${scratch}/main-b2-installed-plan-aee-01/plan.original.json`);
  assert.equal(hash(planBytes), "7ac850a10c253e1aefbd8de1543ac70929718b38c53e45b4bf16315fb0c3907f");
  const plan = JSON.parse(planBytes);
  assert.equal(plan.lintPaths.length, 364);
  for (const file of Object.keys(freeze.source)) assert(plan.lintPaths.includes(file), `SOURCE_NOT_IN_PRIOR_SELECTION:${file}`);
  args = ["node_modules/eslint/bin/eslint.js", "--max-warnings=0", ...plan.lintPaths];
  assert(args.reduce((count, item) => count + item.length + 3, 0) < 28000);
  fs.writeFileSync(`${scratch}/lint-product-fix-01/lint-selection.original.json`, JSON.stringify({ documentClass: "engineering-evidence", at: new Date().toISOString(), sourceFreezeSha256: hash(freezeBytes), priorPlanSha256: hash(planBytes), selectedFiles: plan.lintPaths, maximumWarnings: 0, qualification: false }, null, 2) + "\n", { flag: "wx" });

for (const [file, expected] of Object.entries(freeze.source)) assert.equal(hash(fs.readFileSync(file)), expected, `SOURCE_CHANGED_DURING_CHECK:${file}`);
