import json,subprocess,re
from pathlib import Path
root=Path(r'D:\CodexTemp\parallax-integration-20261019-01a15554\review-cfc5b44d'); out=Path(r'D:\CodexTemp\parallax-integration-20261019-01a15554\independent-review-cfc5b44d')
p=json.loads((root/'packet.json').read_text(encoding='utf-8-sig')); ff=r'C:\Users\kkids\AppData\Local\ms-playwright\ffmpeg-1011\ffmpeg-win64.exe'; rows=[]
for i,m in enumerate(p['media'],1):
 if not m['relative'].endswith('.webm'):continue
 r=subprocess.run([ff,'-i',m['file']],capture_output=True,text=True)
 match=re.search(r'Duration: (\d+):(\d+):(\d+\.\d+)',r.stderr)
