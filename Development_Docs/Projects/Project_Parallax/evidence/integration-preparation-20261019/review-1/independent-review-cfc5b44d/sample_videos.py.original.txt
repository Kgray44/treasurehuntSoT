import json,subprocess
from pathlib import Path
from PIL import Image,ImageDraw
out=Path(r'D:\CodexTemp\parallax-integration-20261019-01a15554\independent-review-cfc5b44d'); vids=json.loads((out/'recordings.json').read_text()); ff=r'C:\Users\kkids\AppData\Local\ms-playwright\ffmpeg-1011\ffmpeg-win64.exe'
ids=['M030','M031','M033','M038','M046','M055','M057','M064','M066','M067','M070','M074','M075','M085','M087','M088','M091','M097','M098','M100']
meta=[]
for v in vids:
 if v['id'] not in ids:continue
 dest=out/'frames'/v['id'];dest.mkdir(parents=True,exist_ok=True); frames=[]
 for k in range(8):
  t=min(max(v['duration']*k/8,0),v['duration']-0.04);f=dest/f'{k:02}-{t:.2f}.png'
  r=subprocess.run([ff,'-loglevel','error','-ss',str(t),'-i',v['file'],'-frames:v','1','-y',str(f)],capture_output=True,text=True)
  if r.returncode!=0 or not f.exists():raise RuntimeError(r.stderr)
  frames.append({'time':round(t,3),'file':str(f)})
 sheet=Image.new('RGB',(2000,820),'#e7eaf0');d=ImageDraw.Draw(sheet)
 for k,fr in enumerate(frames):
  im=Image.open(fr['file']).convert('RGB');im.thumbnail((496,375));x=(k%4)*500;y=(k//4)*410;sheet.paste(im,(x+(500-im.width)//2,y));d.text((x+8,y+379),f"{v['id']} t={fr['time']:.2f}s",fill='black')
 sheet.save(out/f"video-{v['id']}.jpg",quality=93)
 meta.append({**v,'sampleMethod':'Eight decoded native-size frames at duration fractions 0/8 through 7/8; temporal sampling, not real-time playback','frames':frames})
(out/'sampled-recordings.json').write_text(json.dumps(meta,indent=2))
print(f'Extracted {len(meta)} videos / {sum(len(v["frames"]) for v in meta)} native frames')
